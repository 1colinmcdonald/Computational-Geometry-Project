cmake_minimum_required(VERSION 3.26)
cmake_policy(SET CMP0167 NEW)
project(proj LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(CGAL REQUIRED)

add_executable(benchmark benchmark.cpp)
target_precompile_headers(points PRIVATE benchmark.h)

# Works if CGAL provides imported targets
if(TARGET CGAL::CGAL)
  target_link_libraries(points PRIVATE CGAL::CGAL)
else()
  # Fallback for older CGAL CMake configs
  include(${CGAL_USE_FILE})
  target_link_libraries(points PRIVATE ${CGAL_LIBRARIES} ${CGAL_3RD_PARTY_LIBRARIES})
endif()
